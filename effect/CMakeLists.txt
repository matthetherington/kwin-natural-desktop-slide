find_package(Qt6 REQUIRED COMPONENTS DBus Quick Widgets)
find_package(KWin REQUIRED GLOBAL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/compat/kwin-versions.json" version_policy)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/compat/kwin-versions.json")
string(JSON version_count LENGTH "${version_policy}" sourceBuildVersions)
math(EXPR last_version "${version_count} - 1")
set(reviewed_versions)
foreach(index RANGE ${last_version})
    string(JSON reviewed GET "${version_policy}" sourceBuildVersions ${index})
    list(APPEND reviewed_versions "${reviewed}")
endforeach()
if(NOT KWin_VERSION IN_LIST reviewed_versions)
    message(FATAL_ERROR "KWin ${KWin_VERSION} is not source-reviewed. Supported source builds: ${reviewed_versions}")
endif()
message(STATUS "Building for KWin ${KWin_VERSION}; runtime must match this exact build version")
string(JSON tested_count LENGTH "${version_policy}" runtimeTestedVersions)
math(EXPR last_tested "${tested_count} - 1")
set(runtime_tested_versions)
foreach(index RANGE ${last_tested})
    string(JSON tested GET "${version_policy}" runtimeTestedVersions ${index})
    list(APPEND runtime_tested_versions "${tested}")
endforeach()
if(NOT KWin_VERSION IN_LIST runtime_tested_versions)
    message(WARNING "KWin ${KWin_VERSION} is a source-reviewed port, not runtime-validated. Save work and use isolated compositor tests before activation.")
endif()
set(NDS_KWIN_VERSION "${KWin_VERSION}" PARENT_SCOPE)
set(CMAKE_AUTOMOC ON)
list(APPEND CMAKE_AUTOMOC_MACRO_NAMES KWIN_EFFECT_FACTORY_SUPPORTED_ENABLED)

if(NOT DESKTOP_SLIDE_RENDERER_ID MATCHES "^kwin_natural_desktop_slide(_dev[a-zA-Z0-9]+)?$")
    message(FATAL_ERROR "Invalid DESKTOP_SLIDE_RENDERER_ID")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/metadata.json" effect_metadata)
string(JSON effect_metadata SET "${effect_metadata}" KPlugin Id "\"${DESKTOP_SLIDE_RENDERER_ID}\"")
if(TARGET kwin_natural_desktop_slide_config)
    string(JSON effect_metadata SET "${effect_metadata}" X-KDE-ConfigModule "\"kwin_natural_desktop_slide_config\"")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/natural_desktop_slide_metadata.json" CONTENT "${effect_metadata}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/renderer-plugin-id.txt" CONTENT "${DESKTOP_SLIDE_RENDERER_ID}\n")
add_library(kwin_natural_desktop_slide MODULE main.cpp natural_desktop_slide.cpp natural_desktop_slide.h)
target_include_directories(kwin_natural_desktop_slide PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(kwin_natural_desktop_slide PRIVATE panel_policy gesture_settings KWin::kwin)
target_compile_definitions(kwin_natural_desktop_slide PRIVATE NDS_KWIN_BUILD_VERSION="${KWin_VERSION}")
target_compile_options(kwin_natural_desktop_slide PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(kwin_natural_desktop_slide PRIVATE -Wl,--no-undefined)
set_target_properties(kwin_natural_desktop_slide PROPERTIES PREFIX "" OUTPUT_NAME "${DESKTOP_SLIDE_RENDERER_ID}"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
# Use the same non-executable plugin mode as the source installer, independent
# of the distribution's CMAKE_INSTALL_SO_NO_EXE default.
install(TARGETS kwin_natural_desktop_slide
    LIBRARY DESTINATION "${KDE_INSTALL_PLUGINDIR}/kwin/effects/plugins"
    PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ COMPONENT effect)
set(install_artifacts "{\"role\":\"renderer\",\"source\":\"$<TARGET_FILE:kwin_natural_desktop_slide>\",\"name\":\"${DESKTOP_SLIDE_RENDERER_ID}.so\"}")
if(TARGET kwin_natural_desktop_slide_config)
    string(APPEND install_artifacts ",{\"role\":\"settings\",\"source\":\"$<TARGET_FILE:kwin_natural_desktop_slide_config>\",\"name\":\"kwin_natural_desktop_slide_config.so\"}")
endif()

    install(TARGETS kwin_natural_desktop_slide_controller

if(BUILD_CONFIGURATOR)
    string(APPEND install_artifacts ",{\"role\":\"configurator\",\"source\":\"$<TARGET_FILE:natural-desktop-slide-config>\",\"name\":\"natural-desktop-slide-config\"}")
    string(APPEND install_artifacts ",{\"role\":\"launcher\",\"source\":\"${PROJECT_SOURCE_DIR}/packaging/natural-desktop-slide.desktop\",\"name\":\"natural-desktop-slide.desktop\"}")

file(GENERATE OUTPUT "${PROJECT_BINARY_DIR}/install-layout-$<CONFIG>.json" CONTENT
    "{\"schemaVersion\":1,\"kwinVersion\":\"${KWin_VERSION}\",\"integrationDriver\":$<IF:$<BOOL:${ENABLE_INTEGRATION_DRIVER}>,true,false>,\"pluginDirectory\":\"${KDE_INSTALL_FULL_PLUGINDIR}/kwin/effects\",\"binaryDirectory\":\"${KDE_INSTALL_FULL_BINDIR}\",\"applicationDirectory\":\"${KDE_INSTALL_FULL_APPDIR}\",\"artifacts\":[${install_artifacts}]}\n")
